# CMakeLists.txt 配置示例
# 多网配额查询功能的CMake配置

cmake_minimum_required(VERSION 3.16)
project(NetworkBoostQuotaDemo)

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)

# 添加可执行文件
add_executable(entry example_multipath_quota.cpp)

# 配置头文件路径
target_include_directories(entry PUBLIC 
    ${HMOS_SDK_NATIVE}/sysroot/usr/include
)

# 配置动态库路径
target_link_directories(entry PUBLIC 
    ${HMOS_SDK_NATIVE}/sysroot/usr/lib/aarch64-linux-ohos
)

# 链接Network Boost动态库
target_link_libraries(entry PUBLIC 
    libnetwork_boost.so
)

# 编译选项
target_compile_options(entry PRIVATE 
    -Wall
    -Wextra
    -O2
)

# 输出目录设置
set_target_properties(entry PROPERTIES 
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
