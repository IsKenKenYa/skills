# CMakeLists.txt 配置示例

# 多网建议监听功能需要链接 libnetwork_boost.so 动态库
# 请根据实际开发环境调整 HMOS_SDK_NATIVE 路径

cmake_minimum_required(VERSION 3.4.1)
project(NetworkBoostMultiPathRecommendationExample)

# 设置 HarmonyOS SDK Native 路径
# 实际开发中需要替换为真实的 SDK 路径
# 例如: set(HMOS_SDK_NATIVE "/path/to/HarmonyOS/Sdk/10/native")
set(HMOS_SDK_NATIVE "${HMOS_SDK_NATIVE}")

# 添加头文件路径
# 包含 NetworkBoostKit 相关头文件
target_include_directories(entry PUBLIC 
    ${HMOS_SDK_NATIVE}/sysroot/usr/include
)

# 添加动态库路径
# 指定 libnetwork_boost.so 的路径
target_link_directories(entry PUBLIC 
    ${HMOS_SDK_NATIVE}/sysroot/usr/lib/aarch64-linux-ohos
)

# 链接动态库
# 必须链接 libnetwork_boost.so 才能使用 Network Boost Kit API
target_link_libraries(entry PUBLIC 
    libnetwork_boost.so
    # 可能还需要链接其他依赖库,根据实际需求添加
)

# 设置编译选项
# 建议启用 C++11 或更高标准
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置编译目标架构
# HarmonyOS Native 开发通常针对 aarch64-linux-ohos
set(CMAKE_SYSTEM_NAME Linux)
set(CMAKE_SYSTEM_PROCESSOR aarch64)

# 添加源文件
# 根据实际项目结构调整
add_executable(entry 
    src/main.cpp
    # 添加其他源文件
)

# 注意事项:
# 1. HMOS_SDK_NATIVE 路径必须正确配置
# 2. libnetwork_boost.so 必须存在于指定路径
# 3. 需要在 module.json5 中配置权限:
#    - ohos.permission.LINKTURBO (受限 ACL 权限)
# 4. 需要通过 AGC 申请并配置受限 ACL 权限签名
# 5. API 版本要求: 6.0.2(22) 及以上
