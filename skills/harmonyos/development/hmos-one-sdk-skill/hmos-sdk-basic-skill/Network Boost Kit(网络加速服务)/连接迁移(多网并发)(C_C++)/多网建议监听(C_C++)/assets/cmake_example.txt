cmake_minimum_required(VERSION 3.10)
project(networkboost_multipath_recommendation_example)

# 设置C++标准
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 添加Network Boost Kit依赖
find_library(network_boost_lib libnetwork_boost.so)

# 添加头文件路径（根据实际SDK路径调整）
target_include_directories(${PROJECT_NAME}
    PRIVATE
    ${OHOS_SDK_PATH}/native/sysroot/usr/include
)

# 添加可执行文件
add_executable(multipath_recommendation_listener_example
    multipath_recommendation_listener_example.cpp
)

# 接库文件
target_link_libraries(multipath_recommendation_listener_example
    ${network_boost_lib}
)

# 设置输出路径
set_target_properties(multipath_recommendation_listener_example
    PROPERTIES
    OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/bin"
)

# 打印配置信息
message(STATUS "Network Boost Library: ${network_boost_lib}")
message(STATUS "Output Directory: ${CMAKE_CURRENT_BINARY_DIR}/bin")
